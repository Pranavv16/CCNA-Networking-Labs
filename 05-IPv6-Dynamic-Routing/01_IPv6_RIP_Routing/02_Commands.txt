
==================================================
IPv6 RIP ROUTING LAB - CONFIGURATION COMMANDS
==================================================

------------------------------
ROUTER 1 (R1)
------------------------------

enable
configure terminal
ipv6 unicast-routing

interface fastethernet 0/0
ipv6 address FD00::1/64
ipv6 rip R1 enable
no shutdown
exit

interface fastethernet 0/1
ipv6 address 2001::1/64
ipv6 rip R1 enable
no shutdown
exit

ipv6 router rip R1
int f0/0
ipv6 rip R1 enable
int f0/1
ipv6 rip R1 enable
exit
