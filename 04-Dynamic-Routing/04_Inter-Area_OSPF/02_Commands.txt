============================================================
ROUTER 1 (R1) CONFIGURATION
============================================================

enable
configure terminal

interface FastEthernet0/0
ip address 170.170.0.1 255.255.255.252
no shutdown
exit

interface FastEthernet0/1
ip address 192.168.1.1 255.255.255.0
no shutdown
exit

router ospf 1
network 192.168.1.0 0.0.0.255 area 100
network 170.170.0.0 0.0.0.3 area 100
exit

end
